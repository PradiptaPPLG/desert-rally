extends VehicleBody3D

# ==============================================================================
# GODOT 4.x DESERT & COASTAL BEACH DRIFT CONTROLLER
# ==============================================================================
@export var max_engine_force: float = 680.0
@export var max_steering_angle: float = 0.46
@export var steering_speed: float = 6.5
@export var normal_rear_grip: float = 2.6
@export var drift_rear_grip: float = 0.82

@onready var wheel_fl: VehicleWheel3D = $WheelFL
@onready var wheel_fr: VehicleWheel3D = $WheelFR
@onready var wheel_rl: VehicleWheel3D = $WheelRL
@onready var wheel_rr: VehicleWheel3D = $WheelRR
@onready var sand_particles: GPUParticles3D = $SandDriftParticles

func _ready() -> void:
    Input.set_mouse_mode(Input.MOUSE_MODE_HIDDEN)

func _physics_process(delta: float) -> void:
    var throttle_in := Input.get_axis("move_back", "move_forward")
    var steer_in := Input.get_axis("move_right", "move_left")
    var is_drifting := Input.is_action_pressed("handbrake")
    var is_boosting := Input.is_action_pressed("turbo")

    var boost_mult := 1.45 if is_boosting else 1.0
    engine_force = throttle_in * max_engine_force * boost_mult
    steering = move_toward(steering, steer_in * max_steering_angle, steering_speed * delta)

    if is_drifting:
        wheel_rl.wheel_friction_slip = drift_rear_grip
        wheel_rr.wheel_friction_slip = drift_rear_grip
        brake = 2.5
    else:
        wheel_rl.wheel_friction_slip = normal_rear_grip
        wheel_rr.wheel_friction_slip = normal_rear_grip
        brake = 0.0

    var lateral_vel := absf(global_transform.basis.x.dot(linear_velocity))
    sand_particles.emitting = is_drifting or lateral_vel > 3.2
